python - <<'PY'
from pathlib import Path
from fnmatch import fnmatch
from zipfile import ZipFile, ZIP_DEFLATED
from datetime import datetime, timezone

root = Path("evaluation/results/paper-day-v2")
if not root.is_dir():
    raise SystemExit(f"Missing results directory: {root}")

keep = [
    "*request_report.csv", "*job_report.csv",
    "*transfer_overhead_report.csv", "*verification_report.json",
    "*resource_usage.csv", "*stage_transitions.csv",
    "*admission_profile_before.json", "*resource_profile.json",
    "*/snapshot/*.json",
    "*/traffic/*.json", "*/traffic/*.jsonl",
    "*/checks/*.json",
    "*/config/coordinator_*.json", "*/config/node_*.json",
    "*plan.json", "*metadata.json", "*environment.json",
    "*run_status.json", "*setup_times.json", "*setup_source.json",
    "*DONE.json", "*FAILED.json",
    "*coordinator.log", "*node.log", "*node_*.log",
    "*prover_process.log",
]
skip_dirs = {"cache", "trial-setup", "tmp"}

stamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S%fZ")
output = Path(f"paper-day-v2-analysis-{stamp}.zip")
count = 0

with ZipFile(output, "x", ZIP_DEFLATED, compresslevel=1) as archive:
    for path in sorted(root.rglob("*")):
        relative = path.relative_to(root)
        if skip_dirs.intersection(relative.parts[:-1]):
            continue
        if any(parent.is_symlink() for parent in path.parents):
            continue
        if path.is_symlink() or not path.is_file():
            continue
        if not any(fnmatch(relative.as_posix(), p) for p in keep):
            continue
        archive.write(path, arcname=f"local-analysis/{relative.as_posix()}")
        count += 1

Path("analysis-zip-path.txt").write_text(str(output))
print(f"Created {output}: {count} files, "
      f"{output.stat().st_size / 1024**2:.1f} MiB")
PY

ANALYSIS_ZIP="$(cat analysis-zip-path.txt)"
